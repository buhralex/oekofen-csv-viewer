**buildChartData() function** — converts AppState.dataModel into uPlot format:

```javascript
function buildChartData(dataModel, seriesNames) {
  // uPlot data format: [ timestamps[], col1[], col2[], ... ]
  // First element is always the shared X-axis timestamps array.
  const timestamps = dataModel.timestamps;

  // Resolve columns by rawName — skip any that don't exist in the data model
  const resolvedCols = [];
  for (const rawName of seriesNames) {
    const col = dataModel.columns.find(c => c.rawName === rawName);
    if (col) resolvedCols.push(col);
    else console.warn('[buildChartData] rawName not found in dataModel:', rawName);
  }

  // Build columnar data arrays
  const data = [timestamps];
  for (const col of resolvedCols) {
    // uPlot treats NaN as a gap — convert null to NaN
    const arr = col.data.map(v => (v === null ? NaN : v));
    data.push(arr);
  }

  // Build series opts — first entry is X-axis placeholder (empty object)
  const series = [{}];
  for (const col of resolvedCols) {
    const isBinary = col.type === 'discrete';
    const groupColors = GROUP_COLORS[col.group] || { continuous: '#e0e0e0' };
    const stroke = SERIES_COLOR_OVERRIDES[col.rawName] ||
                   (isBinary ? (groupColors.binary || 'rgba(79,195,247,0.45)') : groupColors.continuous) ||
                   '#e0e0e0';

    if (isBinary) {
      // Binary series: filled band on the right 'binary' axis (fixed 0-1 scale)
      // fill: stroke uses the rgba transparency to allow temperature lines to show through
      // paths.stepped renders as a step function (0 or 1, no interpolation)
      series.push({
        label:    col.displayName,
        scale:    'binary',
        stroke:   stroke,
        fill:     stroke,
        width:    0,          // no line stroke — fill only
        spanGaps: false,
        paths:    uPlot.paths.stepped({ align: 1 }),
        points:   { show: false },
      });
    } else {
      series.push({
        label:    col.displayName,
        scale:    'y',
        stroke:   stroke,
        width:    1.5,
        spanGaps: false,
        points:   { show: false },
      });
    }
  }

  return { data, series, resolvedCols };
}
```

**Note on uPlot.paths.stepped**: This exists on the uPlot global in the IIFE build at v1.6.32. Verify it exists by checking `typeof uPlot.paths` in the console after the page loads. If `uPlot.paths` is undefined, replace `paths: uPlot.paths.stepped({ align: 1 })` with `drawStyle: 2` (uPlot's stepped draw mode constant) as a fallback.

**Step D — createChart() function** — add after destroyChart():

```javascript
function createChart() {
  destroyChart();

  const dataModel = AppState.dataModel;
  const { data, series } = buildChartData(dataModel, DEFAULT_SERIES);

  // Format chart title as DD.MM.YYYY from AppState.fileDate (YYYYMMDD) or first timestamp
  let chartTitle = '';
  if (AppState.fileDate && AppState.fileDate.length === 8) {
    const fd = AppState.fileDate;
    chartTitle = fd.slice(6, 8) + '.' + fd.slice(4, 6) + '.' + fd.slice(0, 4);
  } else if (dataModel.timestamps.length > 0) {
    const d = new Date(dataModel.timestamps[0] * 1000);
    chartTitle = String(d.getUTCDate()).padStart(2, '0') + '.' +
                 String(d.getUTCMonth() + 1).padStart(2, '0') + '.' +
                 d.getUTCFullYear();
  }

  const container = document.getElementById('chart-area');
  const opts = {
    title:  chartTitle,
    width:  container.clientWidth,
    height: container.clientHeight,
    series,

    axes: [
      // X-axis: Unix seconds formatted as HH:MM in UTC
      {
        stroke: '#a0a0b8',
        grid:   { stroke: 'rgba(255,255,255,0.10)', width: 1 },
        ticks:  { stroke: 'rgba(255,255,255,0.10)', width: 1 },
        values: (self, ticks) => ticks.map(t => {
          const d = new Date(t * 1000);
          return String(d.getUTCHours()).padStart(2, '0') + ':' +
                 String(d.getUTCMinutes()).padStart(2, '0');
        }),
      },
      // Left Y-axis: continuous series (temperatures, percentages) — auto-scale
      {
        scale:  'y',
        side:   3,
        stroke: '#a0a0b8',
        grid:   { stroke: 'rgba(255,255,255,0.10)', width: 1 },
        ticks:  { stroke: 'rgba(255,255,255,0.10)', width: 1 },
        values: (self, ticks) => ticks.map(v => v.toFixed(0)),
        size:   45,
      },
      // Right Y-axis: binary/discrete series — fixed 0-1 range
      {
        scale:  'binary',
        side:   1,
        stroke: '#a0a0b8',
        grid:   { show: false },
        ticks:  { show: false },
        values: (self, ticks) => ticks.map(v => (v === 1 ? 'On' : v === 0 ? 'Off' : '')),
        size:   35,
      },
    ],

    scales: {
      x:      {},
      y:      { auto: true },    // auto-scales to visible continuous series
      binary: { range: [0, 1] }, // fixed for discrete 0/1 states
    },

    legend: {
      show: true,
      live: false,  // Phase 3 will handle live cursor value display
    },

    cursor: {
      show: false,  // Phase 3 adds crosshair and tooltip
    },

    select: { show: false },  // Phase 3 adds drag-to-zoom selection

    // Reserve minimap space: chart height fills container (Phase 3 will add minimap below)
    padding: [10, 0, 0, 0],
  };

  AppState.chart = new uPlot(opts, data, container);
  AppState.chartSeries = series;

  setStatus('Chart rendered \u2014 ' + dataModel.rowCount + ' rows, ' + (data.length - 1) + ' series');
}
```

```javascript
async function onFileAccepted(file) {
  setStatus('Loading\u2026');

  try {
    const csvString = await readFileAsText(file);
    const papaResult = parseCSVString(csvString);
    const { fields, rows } = normalizeHeaders(papaResult);
    const dataModel = buildDataModel(fields, rows);

    if (!dataModel.isOekoFEN) {
      showToast("This doesn't look like OekoFEN data \u2014 missing Datum/Zeit columns.", 'warning');
      setStatus('Parse failed: no Datum/Zeit columns.');
      return;
    }

    const hasAT = dataModel.columns.some(c => c.rawName === 'AT [\u00b0C]' || c.rawName.trim() === 'AT');
    if (!hasAT) {
      showToast("This doesn't look like OekoFEN data, but proceeding.", 'warning');
    }

    AppState.dataModel = dataModel;
    AppState.filename = file.name;
    const dateMatch = file.name.match(/(\d{8})/);
    AppState.fileDate = dateMatch ? dateMatch[1] : null;

    // Show app view and data summary
    showAppView(file.name);
    showDataSummary(dataModel, file.name);

    // Render chart (Phase 2)
    createChart();

    const issueText = dataModel.parseIssues > 0
      ? ' \u2014 ' + dataModel.parseIssues + ' rows had parse issues'
      : '';
    setStatus('Loaded ' + dataModel.rowCount + ' rows, ' + dataModel.columns.length + ' columns' + issueText);

  } catch (err) {
    console.error('[onFileAccepted] Error:', err);
    showToast('Failed to load file: ' + err.message, 'error');
    setStatus('Error: ' + err.message);
  }
}
```
