    ```javascript
      // PARS-03: The real OekoFEN CSV has NO BOM and is ISO-8859-1 encoded.
      // Using TextDecoder('windows-1252') correctly handles Latin-1 extended chars
      // (°, ä, ö, ü). UTF-8 would produce mojibake (Â°C instead of °C).
      const arrayBuffer = await file.arrayBuffer();
      return new TextDecoder('windows-1252').decode(arrayBuffer);
    }
    ```

    STEP 3 — Normalize headers (trim + drop empty trailing column):
    ```javascript
    function normalizeHeaders(papaResult) {
      // Trailing semicolon creates a column with key "" or " " — filter it out.
      // Column names also have trailing spaces — trim all.
      const rawFields = papaResult.meta.fields;
      const goodFields = rawFields
        .map(f => f.trim())
        .filter(f => f !== '');

      // Rebuild data array with trimmed keys, dropping the empty column
      const rows = papaResult.data.map(row => {
        const normalized = {};
        rawFields.forEach(rawKey => {
          const trimmed = rawKey.trim();
          if (trimmed !== '') {
            normalized[trimmed] = row[rawKey];
          }
        });
        return normalized;
      });

      return { fields: goodFields, rows };
    }
    ```

    STEP 4 — Parse German decimals:
    ```javascript
    // PARS-02: German locale uses comma as decimal separator.
    // parseFloat("1,4") returns 1 in Chrome (stops at comma).
    // Must replace comma with dot before parsing.
    // Returns null for empty/whitespace strings (gap in chart data).
    function parseGermanFloat(str) {
      if (!str || str.trim() === '') return null;
      const val = parseFloat(str.trim().replace(',', '.'));
      return isNaN(val) ? null : val;
    }

    // Classify whether a raw value is an integer (no comma, no dot, digits only)
    function parseGermanValue(str) {
      if (!str || str.trim() === '') return null;
      const trimmed = str.trim();
      // Integer check: digits only, possibly negative (like -2147483648)
      if (/^-?\d+$/.test(trimmed)) {
        const intVal = parseInt(trimmed, 10);
        return isNaN(intVal) ? null : intVal;
      }
      // Float with German decimal comma
      return parseGermanFloat(trimmed);
    }
    ```

    STEP 5 — Build Unix timestamps from Datum + Zeit columns:
    ```javascript
    // PARS-04: Timezone-safe timestamp reconstruction.
    // Always parse from row's Datum and Zeit values, never from filename.
    // Last row may cross midnight to the next calendar day.
    // Returns Unix seconds (not milliseconds — uPlot uses seconds).
    function buildUnixTimestamp(datumStr, zeitStr) {
      // Datum format: DD.MM.YYYY (e.g., "16.02.2026")
      // Zeit format:  HH:MM:SS  (e.g., "00:02:44")
      const [day, month, year] = datumStr.trim().split('.');
      const [hour, min, sec] = zeitStr.trim().split(':');
      // Date.UTC avoids local timezone offset — result is always UTC
      return Date.UTC(+year, +month - 1, +day, +hour, +min, +sec) / 1000;
    }
    ```

    STEP 6 — Column header parsing and group classification:
    ```javascript
    // PARS-05: Parse unit from column name like "AT [°C]" or "HK1 VL Ist[°C]"
    // Returns { displayName, unit }
    function parseColumnHeader(rawName) {
      // Match optional space before bracket: "AT [°C]" or "HK1 VL Ist[°C]"
      const match = rawName.match(/^(.+?)\s*\[([^\]]+)\]\s*$/);
      if (match) {
        return { displayName: rawName, unit: match[2].trim() };
      }
      return { displayName: rawName, unit: null };
    }

    // Per CONTEXT.md group rules (LOCKED USER DECISION):
    // Boiler: AT, ATakt, KT Ist, KT Soll, BR, Sperrzeit, PE1_BR1
    // Heating Circuit (HK1): starts with HK1
    // Hot Water (WW1): starts with WW1
    // Buffer (PU1): starts with PU1
    // Pellet Unit (PE1): starts with PE1 (kept as one group — NOT split)
    // IGNORED: Fehler1, Fehler2, Fehler3, Zubrp1 Pumpe, '' (empty)
    const IGNORED_COLUMNS = new Set(['Fehler1', 'Fehler2', 'Fehler3', 'Zubrp1 Pumpe']);
    const BINARY_COLUMNS = new Set(['BR', 'Sperrzeit']);
    const BOILER_COLUMNS = new Set(['AT', 'ATakt', 'KT Ist', 'KT Soll', 'BR', 'Sperrzeit', 'PE1_BR1']);

    function classifyColumn(rawName) {
      // Extract just the name part (without unit) for group matching
      const namePart = rawName.match(/^(.+?)\s*(?:\[|$)/)?.[1]?.trim() ?? rawName;

      if (IGNORED_COLUMNS.has(namePart)) return null; // excluded
      if (namePart === 'Datum' || namePart === 'Zeit') return 'timestamp'; // X-axis

      let group;
      if (namePart.startsWith('HK1')) group = 'Heating Circuit (HK1)';
      else if (namePart.startsWith('WW1')) group = 'Hot Water (WW1)';
      else if (namePart.startsWith('PU1')) group = 'Buffer (PU1)';
      else if (namePart.startsWith('PE1')) group = 'Pellet Unit (PE1)';
      else group = 'Boiler'; // AT, ATakt, KT Ist/Soll, BR, Sperrzeit, PE1_BR1, others

      const type = BINARY_COLUMNS.has(namePart) ? 'discrete' : 'continuous';
      return { group, type };
    }
    ```

    STEP 7 — Build the columnar data model:
    ```javascript
    function buildDataModel(fields, rows) {
      const datumField = fields.find(f => f === 'Datum');
      const zeitField = fields.find(f => f === 'Zeit');

      if (!datumField || !zeitField) {
        return { isOekoFEN: false, timestamps: [], columns: [] };
      }

      // Build timestamps array (Unix seconds, UTC)
      const timestamps = rows.map(row =>
        buildUnixTimestamp(row['Datum'], row['Zeit'])
      );

      // Build column descriptors for all non-timestamp, non-ignored columns
      const columns = [];
      let parseIssues = 0;

      for (const rawName of fields) {
        const classification = classifyColumn(rawName);
        if (!classification || classification === 'timestamp') continue;

        const { displayName, unit } = parseColumnHeader(rawName);
        const data = rows.map(row => {
          const val = parseGermanValue(row[rawName]);
          if (val === null && row[rawName] && row[rawName].trim() !== '') {
            parseIssues++;
          }
          return val;
        });

        columns.push({
          rawName,
          displayName,   // same as rawName — German names preserved, no translation (INTF-02)
          unit,
          group: classification.group,
          type: classification.type,
          data,
        });
      }

      return {
        isOekoFEN: true,
        timestamps,
        columns,
        parseIssues,
        rowCount: rows.length,
      };
    }
    ```

    ```javascript
        // Detect non-OekoFEN CSV (has Datum/Zeit but no AT column)

        // Store in AppState

        AppState.filename = file.name;

        // Update status bar
    ```

    STEP 9 — Data summary panel (shown after load):
    ```javascript
    function showDataSummary(dataModel, filename) {
      const summary = document.getElementById('data-summary');
      summary.style.display = 'block';

      // Group breakdown
      const groups = {};
      for (const col of dataModel.columns) {
        groups[col.group] = (groups[col.group] || 0) + 1;
      }
      const groupLines = Object.entries(groups)
        .map(([g, n]) => `${g}: ${n} columns`)
        .join(' | ');

      // Time range
      const ts = dataModel.timestamps;
      const firstDate = ts.length > 0 ? new Date(ts[0] * 1000).toUTCString() : 'N/A';
      const lastDate = ts.length > 0 ? new Date(ts[ts.length - 1] * 1000).toUTCString() : 'N/A';

      summary.innerHTML = `
        &lt;div class="summary-bar"&gt;
          &lt;span&gt;&lt;strong&gt;File:&lt;/strong&gt; ${filename}&lt;/span&gt;
          &lt;span&gt;&lt;strong&gt;Rows:&lt;/strong&gt; ${dataModel.rowCount}&lt;/span&gt;
          &lt;span&gt;&lt;strong&gt;Columns:&lt;/strong&gt; ${dataModel.columns.length}&lt;/span&gt;
          &lt;span&gt;&lt;strong&gt;Time:&lt;/strong&gt; ${firstDate} – ${lastDate}&lt;/span&gt;
          &lt;span&gt;&lt;strong&gt;Groups:&lt;/strong&gt; ${groupLines}&lt;/span&gt;
        &lt;/div&gt;
      `;
    }
    ```
