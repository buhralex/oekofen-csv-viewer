**Debounced resize handler** — calls `AppState.chart.setSize()` when the window is resized. Debounce at 100ms to avoid excessive redraws during continuous resize drag:

```javascript
// ─── Responsive resize (CHRT-03) ──────────────────────────────
let _resizeTimer = null;

function onWindowResize() {
  if (!AppState.chart) return;
  clearTimeout(_resizeTimer);
  _resizeTimer = setTimeout(() => {
    const container = document.getElementById('chart-area');
    AppState.chart.setSize({
      width:  container.clientWidth,
      height: container.clientHeight,
    });
  }, 100);
}

window.addEventListener('resize', onWindowResize);
```

**Performance diagnostic function** — available in the browser console to verify CHRT-04 (toggle under 100ms). This is a diagnostic aid, not called at runtime:

```javascript
// ─── Performance diagnostic (CHRT-04) ─────────────────────────
// Usage: benchmarkSeriesToggle() in browser console after loading a CSV
// Expected: < 100ms per toggle
function benchmarkSeriesToggle() {
  if (!AppState.chart) { console.warn('No chart — load a CSV first'); return; }
  const chart = AppState.chart;
  const N = 20;
  const t0 = performance.now();
  for (let i = 0; i < N; i++) {
    const isShown = chart.series[1] ? chart.series[1].show !== false : true;
    chart.setSeries(1, { show: !isShown });
  }
  const elapsed = performance.now() - t0;
  const perToggle = (elapsed / N).toFixed(2);
  console.log('Benchmark: ' + N + ' toggles in ' + elapsed.toFixed(1) + ' ms (' + perToggle + ' ms/toggle)');
  console.log(parseFloat(perToggle) < 100 ? 'CHRT-04: PASS (< 100ms)' : 'CHRT-04: FAIL (> 100ms)');
  chart.setSeries(1, { show: true }); // restore visibility
}
window.benchmarkSeriesToggle = benchmarkSeriesToggle;
```
