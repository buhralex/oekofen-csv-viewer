**Step 2 — Add `createMinimap()` function** above `createChart()` in the `<script>` block:

```javascript
// ─── Minimap Overview (NAVG-05) ───────────────────────────────
// Secondary uPlot instance at 72px height showing full day with zoom region highlighted.
// Bidirectional sync: main chart's setScale hook calls minimap.updateSelection(),
// minimap's setSelect hook calls AppState.chart.setScale() — infinite loop prevented
// by passing false as second arg to minimap.setSelect().
function createMinimap(data, series) {
  const container = document.getElementById('minimap-area');
  if (!container) return null;

  // Simplified series for minimap: reuse same series opts but with thinner strokes
  // to reduce visual noise in 72px height. Only show a subset of key series.
  // Per research: showing all 9 series at full width produces an indistinct mass.
  // Solution: use the same series array but reduce width to 0.5 for continuous, 0 for binary.
  const minimapSeries = series.map(function(s, i) {
    if (i === 0) return s;   // X-axis placeholder — unchanged
    if (s.scale === 'binary') {
      // Binary (BR burner state): keep as filled band but thinner
      return Object.assign({}, s, { width: 0 });
    }
    // Continuous: thin stroke, no fill
    return Object.assign({}, s, { width: 0.5 });
  });

  const minimapOpts = {
    width:  container.clientWidth,
    height: container.clientHeight,   // 72px set by CSS --minimap-height
    series: minimapSeries,

    legend: { show: false },           // no legend in minimap

    axes: [
      { show: false },                 // hide X axis labels (too small at 72px)
      { show: false },                 // hide left Y axis
      { show: false },                 // hide right binary axis
    ],

    cursor: {
      show:   false,                   // no crosshair in minimap
      drag: {
        setScale: false,               // minimap drag does NOT auto-zoom minimap's own scale
        x: true,
        y: false,
      },
    },

    select: {
      show: true,                      // show the highlighted zoom region overlay
    },

    scales: {
      x:      {},
      y:      { auto: true },
      binary: { range: [0, 1] },
    },

    padding: [2, 0, 2, 0],

    hooks: {
      setSelect: [function(uMinimap) {
        // Fires when user drags in minimap to select a new range
        // Convert pixel selection to timestamps and zoom main chart
        const newMin = uMinimap.posToVal(uMinimap.select.left, 'x');
        const newMax = uMinimap.posToVal(uMinimap.select.left + uMinimap.select.width, 'x');
        if (AppState.chart && newMin < newMax) {
          AppState.chart.setScale('x', { min: newMin, max: newMax });
        }
      }],
    },
  };

  const uMinimap = new uPlot(minimapOpts, data, container);

  // updateSelection: called by AppState.onZoomChange when main chart zoom changes.
  // Draws the highlight overlay showing the current zoom window on the minimap.
  // CRITICAL: pass false as second arg to setSelect to prevent minimap's setSelect
  // hook from firing (which would call AppState.chart.setScale again — infinite loop).
  function updateSelection(minTs, maxTs) {
    if (!uMinimap) return;

    // Clamp to minimap's own x range to avoid setSelect with out-of-bounds values
    const dataMin = uMinimap.data[0][0];
    const dataMax = uMinimap.data[0][uMinimap.data[0].length - 1];
    const clampedMin = Math.max(minTs, dataMin);
    const clampedMax = Math.min(maxTs, dataMax);

    const left   = Math.round(uMinimap.valToPos(clampedMin, 'x'));
    const right  = Math.round(uMinimap.valToPos(clampedMax, 'x'));
    const width  = Math.max(0, right - left);
    // height in CSS pixels (bbox.height is in physical pixels / devicePixelRatio)
    const height = uMinimap.bbox.height / devicePixelRatio;

    uMinimap.setSelect({ left: left, width: width, height: height, top: 0 }, false);
  }

  return { instance: uMinimap, updateSelection: updateSelection };
}
```

**Step 5 — Update `onWindowResize()`** to also resize the minimap:

The current `onWindowResize()` only resizes `AppState.chart`. Add minimap resize inside the `setTimeout` callback, after `AppState.chart.setSize(...)`:

```javascript
function onWindowResize() {
  if (!AppState.chart) return;
  clearTimeout(_resizeTimer);
  _resizeTimer = setTimeout(function() {
    const chartContainer = document.getElementById('chart-area');
    AppState.chart.setSize({
      width:  chartContainer.clientWidth,
      height: chartContainer.clientHeight,
    });
    if (AppState.minimap) {
      const minimapContainer = document.getElementById('minimap-area');
      AppState.minimap.instance.setSize({
        width:  minimapContainer.clientWidth,
        height: minimapContainer.clientHeight,
      });
    }
  }, 100);
}
```

Note: The existing `onWindowResize` uses an arrow function. Convert the inner callback to a regular function or keep arrow syntax — either is fine. The key change is adding the minimap setSize call alongside the chart setSize call.
