```javascript
// ─── Settings (v1.1) ──────────────────────────────────────────
const SETTINGS_KEY = 'oekofen-viewer-settings';
const _settings = { ip: '', port: '4321', password: '' };

function loadSettings() {
  try {
    const raw = localStorage.getItem(SETTINGS_KEY);
    if (!raw) return;
    const parsed = JSON.parse(raw);
    if (parsed && typeof parsed === 'object') {
      _settings.ip       = parsed.ip       || '';
      _settings.port     = parsed.port     || '4321';
      _settings.password = parsed.password || '';
    }
  } catch (e) {
    console.warn('[loadSettings] failed:', e.name);
  }
}

function saveSettings(ip, port, password) {
  _settings.ip       = ip;
  _settings.port     = port || '4321';
  _settings.password = password;
  try {
    localStorage.setItem(SETTINGS_KEY, JSON.stringify({
      ip:       _settings.ip,
      port:     _settings.port,
      password: _settings.password,
    }));
  } catch (e) {
    console.warn('[saveSettings] localStorage write failed:', e.name);
  }
}
```

Then at the bottom of the script, in the Init section (just before `setStatus('Ready...')`), add the call:
```javascript

```

- Default port is `'4321'` (string, not number — consistent with how it will be used in URL construction)

**Step B — Add onboarding JS logic in the Settings section (after loadSettings/saveSettings):**

```javascript
const ONBD_KEY = 'oekofen-viewer-onboarding';

function isOnboardingDismissed() {
  try {
    return localStorage.getItem(ONBD_KEY) === 'dismissed';
  } catch (e) {
    return false;
  }
}

function dismissOnboarding() {
  try {
    localStorage.setItem(ONBD_KEY, 'dismissed');
  } catch (e) {
    console.warn('[dismissOnboarding] localStorage write failed:', e.name);
  }
  const el = document.getElementById('onboarding-prompt');
  if (el) el.style.display = 'none';
}

function initOnboardingPrompt() {
  // Show only if: no settings saved AND not previously dismissed
  if (_settings.ip || isOnboardingDismissed()) return;
  const prompt = document.getElementById('onboarding-prompt');
  if (prompt) prompt.style.display = '';

  const connectBtn = document.getElementById('onboarding-connect-btn');
  if (connectBtn) {
    connectBtn.addEventListener('click', () => {
      dismissOnboarding();
      openSettingsModal(); // wired in plan 05-02
    });
  }

  const dismissBtn = document.getElementById('onboarding-dismiss-btn');
  if (dismissBtn) {
    dismissBtn.addEventListener('click', () => {
      dismissOnboarding();
    });
  }
}
```
