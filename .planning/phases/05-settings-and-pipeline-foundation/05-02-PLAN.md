**Step C — Add openSettingsModal() and closeSettingsModal() in the script:**

Place these functions in the script after the picker modal functions (search for `function openPickerModal()` — place the settings modal functions just before the event wiring section, around line 2515).

```javascript
// ─── Settings Modal (v1.1) ────────────────────────────────────

function closeSettingsModal() {
  const el = document.getElementById('settings-modal');
  if (el) el.remove();
  document.removeEventListener('keydown', _settingsEscHandler);
}

let _settingsEscHandler = null;

function openSettingsModal() {
  if (document.getElementById('settings-modal')) return; // prevent double-open

  // ── Backdrop ────────────────────────────────────────────────
  const backdrop = document.createElement('div');
  backdrop.id = 'settings-modal';
  backdrop.style.cssText = [
    'position:fixed; inset:0; z-index:200;',
    'background:rgba(22,25,43,0.35);',
    'display:flex; align-items:center; justify-content:center;',
  ].join('');
  backdrop.addEventListener('click', (e) => {
    if (e.target === backdrop) closeSettingsModal();
  });

  // ── Dialog ──────────────────────────────────────────────────
  const dialog = document.createElement('div');
  dialog.style.cssText = [
    'background:var(--bg-l1); border-radius:var(--r-lg);',
    'box-shadow:0 8px 32px rgba(22,25,43,0.18);',
    'width:360px; max-width:calc(100vw - 32px);',
    'font-family:var(--font);',
  ].join('');

  // Header
  const header = document.createElement('div');
  header.style.cssText = 'padding:16px 20px 0; font-size:15px; font-weight:600; color:var(--text-1);';
  header.textContent = 'Heater Connection Settings';
  dialog.appendChild(header);

  // Body
  const body = document.createElement('div');
  body.style.cssText = 'padding:16px 20px;';

  function makeField(labelText, inputId, placeholder, currentValue) {
    const wrap = document.createElement('div');
    wrap.style.cssText = 'margin-bottom:14px;';
    const lbl = document.createElement('label');
    lbl.htmlFor = inputId;
    lbl.style.cssText = 'display:block; font-size:12px; font-weight:500; color:var(--text-2); margin-bottom:4px;';
    lbl.textContent = labelText;
    const inp = document.createElement('input');
    inp.type = 'text';
    inp.id = inputId;
    inp.placeholder = placeholder;
    inp.value = currentValue;
    inp.style.cssText = [
      'width:100%; padding:7px 10px;',
      'border:1px solid var(--sep-strong); border-radius:var(--r-sm);',
      'font-size:13px; font-family:var(--font); color:var(--text-1);',
      'background:var(--bg-l2); outline:none;',
      'transition:border-color 0.15s;',
    ].join('');
    inp.addEventListener('focus', () => inp.style.borderColor = 'var(--blue)');
    inp.addEventListener('blur',  () => inp.style.borderColor = 'var(--sep-strong)');
    wrap.appendChild(lbl);
    wrap.appendChild(inp);
    return wrap;
  }

  body.appendChild(makeField('IP Address', 'settings-ip',   '192.168.1.xxx', _settings.ip));
  body.appendChild(makeField('Port',       'settings-port', '4321',          _settings.port));

  // Password field with note (not type="password" — API token shown in URL, masking impedes verification)
  const pwWrap = makeField('API Password', 'settings-password', 'your-api-password', _settings.password);
  const pwNote = document.createElement('p');
  pwNote.style.cssText = 'font-size:11px; color:var(--text-3); margin-top:3px;';
  pwNote.textContent = 'Stored locally in your browser. Not transmitted to any server.';
  pwWrap.appendChild(pwNote);
  body.appendChild(pwWrap);

  dialog.appendChild(body);

  // Footer
  const footer = document.createElement('div');
  footer.style.cssText = [
    'padding:12px 20px 16px;',
    'display:flex; justify-content:flex-end; gap:8px;',
    'border-top:1px solid var(--sep);',
  ].join('');

  const cancelBtn = document.createElement('button');
  cancelBtn.textContent = 'Cancel';
  cancelBtn.style.cssText = [
    'padding:7px 16px; border-radius:var(--r-sm); border:1px solid var(--sep-strong);',
    'background:none; color:var(--text-2); font-size:13px; font-family:var(--font);',
    'cursor:pointer;',
  ].join('');
  cancelBtn.addEventListener('click', closeSettingsModal);

  const saveBtn = document.createElement('button');
  saveBtn.textContent = 'Save';
  saveBtn.style.cssText = [
    'padding:7px 16px; border-radius:var(--r-sm); border:none;',
    'background:var(--blue); color:#fff; font-size:13px; font-family:var(--font);',
    'font-weight:500; cursor:pointer;',
  ].join('');
  saveBtn.addEventListener('click', () => {
    const ip  = document.getElementById('settings-ip').value.trim();
    const port = document.getElementById('settings-port').value.trim();
    const pass = document.getElementById('settings-password').value.trim();
    saveSettings(ip, port, pass);
    closeSettingsModal();
    showToast('Settings saved.', 'info', 2000);
  });

  footer.appendChild(cancelBtn);
  footer.appendChild(saveBtn);
  dialog.appendChild(footer);
  backdrop.appendChild(dialog);
  document.body.appendChild(backdrop);

  // Focus first field
  setTimeout(() => { const el = document.getElementById('settings-ip'); if (el) el.focus(); }, 50);

  // Escape to close
  _settingsEscHandler = (e) => { if (e.key === 'Escape') closeSettingsModal(); };
  document.addEventListener('keydown', _settingsEscHandler);
}
```

- Modal must pre-fill all three fields from `_settings` at open time (reads _settings.ip, _settings.port, _settings.password)
- Cancel must NOT call saveSettings — leave _settings unchanged
- Save button calls `saveSettings(ip, port, pass)` — the saveSettings() function from plan 05-01 handles _settings mutation and localStorage write
- Double-open guard: `if (document.getElementById('settings-modal')) return;`
- Escape closes modal (same pattern as picker modal)
- `_settingsEscHandler` stored as module-level var so it can be removed on close (prevent handler leak)
