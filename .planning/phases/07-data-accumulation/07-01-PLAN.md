Add a self-contained IndexedDB module as an inline `<script>` block in index.html (or appended to the existing JS section), before the closing `</script>` tag of the main script block. The module must expose four async functions at module scope (not inside any function):

```javascript
// DB constants
const HISTORY_DB_NAME = 'oekofen-history';
const HISTORY_DB_VERSION = 1;
const HISTORY_STORE = 'days';

// Open (or upgrade) the IndexedDB database.
// Returns a Promise<IDBDatabase>.
// Store schema: keyPath='date' (string YYYYMMDD), one index not required.
async function openHistoryDb() {
  return new Promise((resolve, reject) => {
    const req = indexedDB.open(HISTORY_DB_NAME, HISTORY_DB_VERSION);
    req.onupgradeneeded = e => {
      const db = e.target.result;
      if (!db.objectStoreNames.contains(HISTORY_STORE)) {
        db.createObjectStore(HISTORY_STORE, { keyPath: 'date' });
      }
    };
    req.onsuccess = e => resolve(e.target.result);
    req.onerror   = e => reject(e.target.error);
  });
}

// Upsert (put) a record. Existing record for same date is replaced.
// date: string YYYYMMDD; csvString: raw decoded CSV text.
async function upsertHistoryDay(date, csvString) {
  if (!date) return;   // no date = cannot key — skip silently
  const db = await openHistoryDb();
  return new Promise((resolve, reject) => {
    const tx = db.transaction(HISTORY_STORE, 'readwrite');
    tx.objectStore(HISTORY_STORE).put({ date, csvString, storedAt: Date.now() });
    tx.oncomplete = resolve;
    tx.onerror    = e => reject(e.target.error);
  });
}

// Return all stored records sorted by date ascending.
async function getAllHistoryDays() {
  const db = await openHistoryDb();
  return new Promise((resolve, reject) => {
    const tx  = db.transaction(HISTORY_STORE, 'readonly');
    const req = tx.objectStore(HISTORY_STORE).getAll();
    req.onsuccess = e => resolve(
      (e.target.result || []).sort((a, b) => a.date.localeCompare(b.date))
    );
    req.onerror = e => reject(e.target.error);
  });
}

// Clear all records.
async function clearHistoryDb() {
  const db = await openHistoryDb();
  return new Promise((resolve, reject) => {
    const tx = db.transaction(HISTORY_STORE, 'readwrite');
    tx.objectStore(HISTORY_STORE).clear();
    tx.oncomplete = resolve;
    tx.onerror    = e => reject(e.target.error);
  });
}
```

Add a `updateHistoryIndicator()` function in the JS section that reads IndexedDB and updates the element:

```javascript
async function updateHistoryIndicator() {
  const el = document.getElementById('history-indicator');
  if (!el) return;
  try {
    const days = await getAllHistoryDays();
    if (days.length === 0) {
      el.textContent = '';
      el.innerHTML   = '';
      return;
    }
    const count = days.length;
    const first = formatHistoryDate(days[0].date);
    const last  = formatHistoryDate(days[days.length - 1].date);
    const range = count === 1 ? first : first + ' \u2013 ' + last;
    // Render as: "12 days stored: 2026-01-14 – 2026-02-25   [Clear]"
    el.innerHTML =
      count + ' day' + (count === 1 ? '' : 's') + ' stored: ' + range +
      '&nbsp;&nbsp;<button id="clear-history-btn" style="background:none; border:none; color:var(--text-3); font-size:11px; cursor:pointer; font-family:var(--font); text-decoration:underline; padding:0;">Clear</button>';
    document.getElementById('clear-history-btn').addEventListener('click', async () => {
      await clearHistoryDb();
      updateHistoryIndicator();
    });
  } catch (e) {
    console.warn('[history] Failed to read IndexedDB for indicator:', e);
  }
}

// Format YYYYMMDD → YYYY-MM-DD for display
function formatHistoryDate(yyyymmdd) {
  if (!yyyymmdd || yyyymmdd.length !== 8) return yyyymmdd;
  return yyyymmdd.slice(0,4) + '-' + yyyymmdd.slice(4,6) + '-' + yyyymmdd.slice(6,8);
}
```
