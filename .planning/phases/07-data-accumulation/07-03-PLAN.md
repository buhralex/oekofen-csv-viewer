```javascript
// DACC-04: On page load, check /history for server-stored CSVs and import
// any dates not already in IndexedDB. Runs silently — no toasts.
// Only runs when app is served via server.py (http: origin). No-op on file://.
async function loadHistoryFromServer() {
  if (window.location.protocol === 'file:') return; // file:// has no server
  try {
    const resp = await fetch('/history', { signal: AbortSignal.timeout(3000) });
    if (!resp.ok) return; // server running but /history not available — skip
    const serverDates = await resp.json(); // string[] of YYYYMMDD
    if (!Array.isArray(serverDates) || serverDates.length === 0) return;

    // Only import dates not already in IndexedDB
    const existing = await getAllHistoryDays();
    const existingSet = new Set(existing.map(r => r.date));
    const toImport = serverDates.filter(d => !existingSet.has(d));
    if (toImport.length === 0) return;

    let imported = 0;
    for (const date of toImport) {
      try {
        const csvResp = await fetch('/history/' + date + '.csv', { signal: AbortSignal.timeout(5000) });
        if (!csvResp.ok) continue;
        const buffer = await csvResp.arrayBuffer();
        const csvString = new TextDecoder('windows-1252').decode(buffer);
        await upsertHistoryDay(date, csvString);
        imported++;
      } catch (e) {
        console.warn('[history] Failed to import ' + date + ':', e);
      }
    }
    if (imported > 0) {
      await updateHistoryIndicator();
      console.log('[history] Imported ' + imported + ' day(s) from server history');
    }
  } catch (e) {
    // Silently swallow — /history may not be available in older server.py versions
    console.debug('[history] loadHistoryFromServer skipped:', e.message);
  }
}
```

Note: `loadHistoryFromServer()` is called without `await` intentionally — it runs async in background so it does not delay initial page render. The indicator updates itself via `updateHistoryIndicator()` when imports complete.

Key behaviors to preserve:
- File:// check at top: no-op when opened as a local file (prevents fetch error in console)
- 3-second timeout on /history fetch: if server.py is an old version without /history, the call times out cleanly
- `console.debug` (not `console.warn`) for the catch: treated as trace-level noise, not a warning
- TextDecoder('windows-1252') is required for the same reason as in fetchCsv() — OekoFEN CSVs are windows-1252 encoded
