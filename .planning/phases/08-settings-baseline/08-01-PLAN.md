function loadBaseline() {
  try {
    const raw = localStorage.getItem(BASELINE_KEY);
    if (!raw) return;
    const parsed = JSON.parse(raw);
    if (parsed && typeof parsed === 'object' && parsed.sections) {
      _baseline = parsed;
    }
  } catch (e) {
    console.warn('[loadBaseline] failed:', e.name);
  }
}

}

Add parseBaselineTxt(rawText) immediately after saveBaseline(). The OekoFEN .txt settings export contains 5 distinct line types that must all be handled:

1. **Separator lines** — 10+ dashes (e.g. `----------`) → reset sub-section stack
2. **Top-level section headers** — indent 0, not a separator, not starting with `Anlagennummer` → new section
3. **File header** — `Anlagennummer ...` at indent 0 → skip
4. **Options list** — content starts with `[` OR current options accumulation has not yet closed `]` → accumulate until `]` found (may span two raw lines)
5. **Tag: line** — content matches `/^Tag:\s+(.+)/` → start of a day-group within the current schedule
6. **Time block** — content matches `/^(\d{2}:\d{2})\s{2,}(\d{2}:\d{2})(?:\s{2,}(.+))?$/` → time block entry
7. **Setting with value** — content has 2+ space gap between key portion and right-side value → setting entry (key path-qualified by sub-section stack)
8. **Sub-section header** — indented content with no right-side value after 2+ spaces → push onto sub-section stack (or pop if indent level decreases)

The parser is indent-aware and stack-based. Sub-section path tracking: compare leading spaces of each line against the previous line's indent level to detect unindent (pop stack) or same/deeper indent (push or stay). Path-qualified keys are built by joining the stack with ` / ` (e.g. `Solares Heizen / Einschalttemperatur`).

Named schedules (e.g. `Heizkreis HK1 Zeit 1`) are entries that appear as sub-section headers containing "Zeit" — these start a new schedule entry rather than a settings sub-section.

Output shape per section:
```
{
  settings: [ { key: string, value: string, options: string[]|null }, ... ],
  schedules: [
    {
      name: string,
      groups: [ { days: string[], blocks: [ { from, to, mode: string|null } ] } ]
    }
  ]
}
```

Parser implementation:

```js
function parseBaselineTxt(rawText) {
  // Returns { filename: null, sectionCount, sections: { name: { settings, schedules } } }
  // Throws Error with descriptive message if zero top-level sections detected.
  const lines = rawText.split(/\r?\n/);
  const sections = {};
  let sectionCount = 0;

  let currentSection = null;   // key into sections{}
  let subStack = [];           // [ { name, indent } ] — path of sub-section labels
  let currentSchedule = null;  // { name, groups }
  let currentGroup = null;     // { days, blocks }

  // Options accumulation state
  let optionsPending = false;  // true while inside a multi-line [ ... ]
  let optionsAccum = '';       // accumulated raw text for current options list
  let lastSettingRef = null;   // reference to the last settings entry (to attach options)

  function indentOf(rawLine) {
    let i = 0;
    while (i < rawLine.length && (rawLine[i] === ' ' || rawLine[i] === '\t')) i++;
    return i;
  }

  function flushOptions() {
    if (!lastSettingRef) return;
    // Parse "[A, B, C]" → ["A", "B", "C"]
    const inner = optionsAccum.replace(/^\[/, '').replace(/\].*$/, '').trim();
    lastSettingRef.options = inner.split(',').map(s => s.trim()).filter(Boolean);
    lastSettingRef = null;
    optionsPending = false;
    optionsAccum = '';
  }

  function subPath(extraLabel) {
    const parts = subStack.map(e => e.name);
    if (extraLabel) parts.push(extraLabel);
    return parts.join(' / ');
  }

  function ensureSection(name) {
    if (!sections[name]) {
      sections[name] = { settings: [], schedules: [] };
      sectionCount++;
    }
  }

  for (const rawLine of lines) {
    const trimmed = rawLine.trimEnd();
    const content = trimmed.trim();
    if (!content) continue;

    const indent = indentOf(trimmed);

    // ── Options continuation ────────────────────────────────────────
    if (optionsPending) {
      optionsAccum += ' ' + content;
      if (content.includes(']')) flushOptions();
      continue;
    }

    // ── Separator line (10+ dashes) ─────────────────────────────────
    if (/^-{10,}/.test(content)) {
      subStack = [];
      currentSchedule = null;
      currentGroup = null;
      continue;
    }

    // ── File header (Anlagennummer at indent 0) ─────────────────────
    if (indent === 0 && /^Anlagennummer\b/.test(content)) continue;

    // ── Top-level section header (indent 0, not a separator) ────────
    if (indent === 0) {
      currentSection = content;
      ensureSection(currentSection);
      subStack = [];
      currentSchedule = null;
      currentGroup = null;
      continue;
    }

    if (!currentSection) continue;

    // ── Tag: line ───────────────────────────────────────────────────
    const tagMatch = content.match(/^Tag:\s+(.+)/);
    if (tagMatch) {
      const days = tagMatch[1].split(/,\s*/).map(d => d.trim()).filter(Boolean);
      currentGroup = { days, blocks: [] };
      if (currentSchedule) currentSchedule.groups.push(currentGroup);
      continue;
    }

    // ── Time block line ─────────────────────────────────────────────
    const timeMatch = content.match(/^(\d{2}:\d{2})\s{2,}(\d{2}:\d{2})(?:\s{2,}(.+))?$/);
    if (timeMatch) {
      const block = {
        from: timeMatch[1],
        to: timeMatch[2],
        mode: timeMatch[3] ? timeMatch[3].trim() : null
      };
      if (currentGroup) {
        currentGroup.blocks.push(block);
      } else if (currentSchedule) {
        // time blocks before any Tag: line — create an unnamed group
        currentGroup = { days: [], blocks: [block] };
        currentSchedule.groups.push(currentGroup);
      }
      continue;
    }

    // ── Adjust sub-section stack for unindent ───────────────────────
    while (subStack.length > 0 && indent <= subStack[subStack.length - 1].indent) {
      subStack.pop();
    }

    // ── Options list start ──────────────────────────────────────────
    if (content.startsWith('[')) {
      optionsAccum = content;
      if (content.includes(']')) {
        flushOptions();
      } else {
        optionsPending = true;
      }
      continue;
    }

    // ── Setting with value (2+ space gap) ──────────────────────────
    const kvMatch = content.match(/^(.+?)\s{2,}(\S.*)$/);
    if (kvMatch) {
      const keyLabel = kvMatch[1].trim();
      const value = kvMatch[2].trim();
      const qualifiedKey = subPath(keyLabel);

      // Detect schedule name entries containing "Zeit"
      if (/\bZeit\b/.test(qualifiedKey) && value === '') {
        // Schedule header with no value — fall through to sub-section logic below
      } else {
        const entry = { key: qualifiedKey, value, options: null };
        sections[currentSection].settings.push(entry);
        lastSettingRef = entry;
        continue;
      }
    }

    // ── Sub-section header (no right-side value, or schedule name) ──
    // Detect schedule entries: labels containing "Zeit" (e.g. "Heizkreis HK1 Zeit 1")
    const scheduleMatch = content.match(/^(.+\bZeit\s+\d+.*)$/);
    if (scheduleMatch) {
      const scheduleName = (currentSection + ' ' + scheduleMatch[1]).trim();
      currentSchedule = { name: scheduleName, groups: [] };
      currentGroup = null;
      sections[currentSection].schedules.push(currentSchedule);
      subStack.push({ name: content, indent });
    } else {
      // Plain sub-section header: push to path stack
      subStack.push({ name: content, indent });
      // If the label itself looks like a schedule (contains "Zeit"), also open a schedule
    }
  }

  if (sectionCount === 0) {
    throw new Error('No OekoFEN settings sections found. Is this the correct .txt export?');
  }

  return { filename: null, sectionCount, sections };
}
```

**JS functions:** Add the following JS functions in the Settings (v1.1) section, after initOnboardingPrompt():

```js
function showBaselinePrompt() {
  const el = document.getElementById('baseline-prompt');
  if (el) el.style.display = '';
}

function hideBaselinePrompt() {
  const el = document.getElementById('baseline-prompt');
  if (el) el.style.display = 'none';
}

function initBaselinePrompt() {
  const fileInput = document.getElementById('baseline-file-input');
  const statusEl = document.getElementById('baseline-prompt-status');
  const skipBtn = document.getElementById('baseline-skip-btn');

  if (fileInput) {
    fileInput.addEventListener('change', (e) => {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = (ev) => {
        try {
          const parsed = parseBaselineTxt(ev.target.result);
          parsed.filename = file.name;
          saveBaseline(parsed);
          if (statusEl) {
            statusEl.style.color = 'var(--text-2)';
            statusEl.textContent = '\u2713 Loaded: ' + file.name + ' (' + parsed.sectionCount + ' sections)';
          }
          // Auto-hide after 1.5s on success
          setTimeout(() => hideBaselinePrompt(), 1500);
        } catch (err) {
          if (statusEl) {
            statusEl.style.color = '#DC2626';
            statusEl.textContent = err.message;
          }
          // Reset file input so same file can be retried
          fileInput.value = '';
        }
      };
      reader.readAsText(file, 'utf-8');
    });
  }

  if (skipBtn) {
    skipBtn.addEventListener('click', () => hideBaselinePrompt());
  }
}
```
