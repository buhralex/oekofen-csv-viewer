```js
function loadBaseline() {
  try {
    const raw = localStorage.getItem(BASELINE_KEY);
    if (!raw) return;
    const parsed = JSON.parse(raw);
    if (parsed && typeof parsed === 'object' && parsed.sections) {
      _baseline = parsed;
    }
  } catch (e) {
    console.warn('[loadBaseline] failed:', e.name);
  }
}
```

```js
function saveBaseline(parsed) {
  _baseline = parsed;

  }
}
```

Add parseBaselineTxt(rawText) immediately after saveBaseline(). The OekoFEN .txt settings export uses a fixed structure:
- Section headers are lines that end with a colon and contain no '=' sign, e.g. "Heizkreis:" or "Warmwasser 1:"
- Key-value lines are in the format "  Key Name  Value" — whitespace-separated with the value being the last token and the key being everything before it after trimming
- Blank lines separate sections
- Lines with '#' prefix are comments — skip them
- Lines that are purely numeric or match no known pattern — skip silently (they appear as standalone values in some sections)

The actual OekoFEN .txt format is:
```
Heizkreis 1:
  Betriebsart                   3
  Raumtemperatur Soll           22.0
  ...

Warmwasser 1:
  Betriebsart                   1
  ...
```

Parser implementation:

```js
function parseBaselineTxt(rawText) {
  // Returns { sections: { sectionName: { key: value, ... }, ... }, filename: null }
  // Throws Error with descriptive message if no OekoFEN sections found
  const lines = rawText.split(/\r?\n/);
  const sections = {};
  let currentSection = null;
  let sectionCount = 0;

  for (const rawLine of lines) {
    const line = rawLine.trimEnd();
    if (!line.trim() || line.trim().startsWith('#')) continue;

    // Section header: non-empty line ending with ':' that contains no '='
    if (/^[^\s=][^=]*:\s*$/.test(line.trim())) {
      currentSection = line.trim().replace(/:$/, '').trim();
      if (!sections[currentSection]) {
        sections[currentSection] = {};
        sectionCount++;
      }
      continue;
    }

    // Key-value line: must be inside a section and have leading whitespace (indented)
    if (currentSection && rawLine.startsWith(' ') || rawLine.startsWith('\t')) {
      const trimmed = line.trim();
      if (!trimmed || trimmed.startsWith('#')) continue;
      // Split on 2+ spaces or tab to separate key from value
      // Key is left side, value is rightmost token
      const match = trimmed.match(/^(.+?)\s{2,}(\S+)\s*$/) || trimmed.match(/^(.+?)\t(\S+)\s*$/);
      if (match) {
        const key = match[1].trim();
        const value = match[2].trim();
        if (key) sections[currentSection][key] = value;
      }
      // Lines that don't match key=value pattern (standalone values) are silently skipped
    }
  }

  if (sectionCount === 0) {
    throw new Error('No OekoFEN settings sections found. Is this the correct .txt export?');
  }

  return { sections, sectionCount };
}
```

**JS functions:** Add the following JS functions in the Settings (v1.1) section, after initOnboardingPrompt():

```js
function showBaselinePrompt() {
  const el = document.getElementById('baseline-prompt');
  if (el) el.style.display = '';
}

function hideBaselinePrompt() {
  const el = document.getElementById('baseline-prompt');
  if (el) el.style.display = 'none';
}

function initBaselinePrompt() {
  const fileInput = document.getElementById('baseline-file-input');
  const statusEl = document.getElementById('baseline-prompt-status');
  const skipBtn = document.getElementById('baseline-skip-btn');

  if (fileInput) {
    fileInput.addEventListener('change', (e) => {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = (ev) => {
        try {
          const parsed = parseBaselineTxt(ev.target.result);
          parsed.filename = file.name;
          saveBaseline(parsed);
          if (statusEl) {
            statusEl.style.color = 'var(--text-2)';
            statusEl.textContent = '\u2713 Loaded: ' + file.name + ' (' + parsed.sectionCount + ' sections)';
          }
          // Auto-hide after 1.5s on success
          setTimeout(() => hideBaselinePrompt(), 1500);
        } catch (err) {
          if (statusEl) {
            statusEl.style.color = '#DC2626';
            statusEl.textContent = err.message;
          }
          // Reset file input so same file can be retried
          fileInput.value = '';
        }
      };
      reader.readAsText(file, 'utf-8');
    });
  }

  if (skipBtn) {
    skipBtn.addEventListener('click', () => hideBaselinePrompt());
  }
}
```
