3. JS — add loadStatsPanel() function in the main script block (before the Init section at the bottom):

    // Phase 9: Statistics Panel (AGGR-01, AGGR-02)
    async function loadStatsPanel() {
      const panel = document.getElementById('stats-panel');
      panel.innerHTML = '<div class="stats-empty">Loading statistics...</div>';

      // file:// fallback: /stats endpoint is unavailable
      if (window.location.protocol === 'file:') {
        panel.innerHTML = '<div class="stats-empty">Statistics require server.py.<br>Double-click start.bat or run: python server.py</div>';
        return;
      }

      let statsData;
      try {
        const resp = await fetch('/stats', { signal: AbortSignal.timeout(5000) });
        if (!resp.ok) throw new Error('HTTP ' + resp.status);
        statsData = await resp.json();
      } catch (e) {
        panel.innerHTML = '<div class="stats-empty">Could not load statistics. Is server.py running?</div>';
        return;
      }

      if (!statsData.days || statsData.days.length === 0) {
        panel.innerHTML = '<div class="stats-empty">No statistics yet — fetch or upload at least one day\'s CSV to begin.</div>';
        return;
      }

      const days = statsData.days;
      const trend = statsData.trend;

      // Summary section
      const totalStarts = days.reduce((s, d) => s + (d.starts || 0), 0);
      const totalRuntime = days.reduce((s, d) => s + (d.runtime_minutes || 0), 0);
      const runtimeHours = (totalRuntime / 60).toFixed(1);
      const trendLabel = trend.label || 'N/A';
      const periodStart = formatHistoryDate(days[0].date);
      const periodEnd   = formatHistoryDate(days[days.length - 1].date);
      const periodStr   = days.length === 1 ? periodStart : periodStart + ' - ' + periodEnd;

      const summaryHtml = '' +
        '<div class="stats-summary">' +
        '<div class="stats-summary-item"><span class="stats-summary-label">Period</span><span class="stats-summary-value" style="font-size:13px;">' + periodStr + '</span></div>' +
        '<div class="stats-summary-item"><span class="stats-summary-label">Days stored</span><span class="stats-summary-value">' + statsData.total_days + '</span></div>' +
        '<div class="stats-summary-item"><span class="stats-summary-label">Total starts</span><span class="stats-summary-value">' + totalStarts + '</span></div>' +
        '<div class="stats-summary-item"><span class="stats-summary-label">Total runtime</span><span class="stats-summary-value">' + runtimeHours + ' h</span></div>' +
        '<div class="stats-summary-item"><span class="stats-summary-label">Start trend</span><span class="stats-summary-value" style="font-size:14px;">' + trendLabel + '</span></div>' +
        '</div>';

      // Per-day table
      const fmtNum = (v, dec) => { dec = dec === undefined ? 1 : dec; return (v !== null && v !== undefined) ? Number(v).toFixed(dec) : '-'; };
      const rowsHtml = days.map(function(d) {
        const dateLabel = formatHistoryDate(d.date);
        const partialBadge = d.is_partial ? '<span class="stats-partial-badge">*partial</span>' : '';
        const startsVal = (d.starts !== null && d.starts !== undefined) ? d.starts : '-';
        const runtimeVal = d.runtime_minutes ? fmtNum(d.runtime_minutes / 60) + ' h' : '-';
        const pelletVal = (d.pellet_kg !== null && d.pellet_kg !== undefined) ? fmtNum(d.pellet_kg) + ' kg' : 'N/A';
        const outdoorVal = (d.avg_outdoor_temp !== null && d.avg_outdoor_temp !== undefined) ? fmtNum(d.avg_outdoor_temp) + ' C' : '-';
        const deltaVal = fmtNum(d.flow_return_delta) + ' C';
        return '<tr class="stats-row" data-date="' + d.date + '">' +
               '<td>' + dateLabel + partialBadge + '</td>' +
               '<td>' + startsVal + '</td>' +
               '<td>' + runtimeVal + '</td>' +
               '<td>' + pelletVal + '</td>' +
               '<td>' + outdoorVal + '</td>' +
               '<td>' + deltaVal + '</td>' +
               '</tr>';
      }).join('');

      const tableHtml = '<div class="stats-table-wrap"><table class="stats-table">' +
        '<thead><tr><th>Date</th><th>Starts</th><th>Runtime</th><th>Consumption</th><th>Avg Outdoor</th><th>Flow/Return Delta</th></tr></thead>' +
        '<tbody>' + rowsHtml + '</tbody></table></div>' +
        '<div style="margin-top:8px; font-size:11px; color:var(--text-3);">* Partial day (less than 20h of data) — excluded from trend calculation</div>';

      panel.innerHTML = summaryHtml + tableHtml;

      // Row click: load that day from IndexedDB into chart
      panel.querySelectorAll('tr.stats-row').forEach(function(row) {
        row.addEventListener('click', async function() {
          const date = row.dataset.date;
          try {
            const allDays = await getAllHistoryDays();
            const record = allDays.find(function(r) { return r.date === date; });
            if (!record) {
              showToast('Day ' + formatHistoryDate(date) + ' not in local history — fetch it first.', 'warning');
              return;
            }
            showStatsPanel(false);
            await onCsvStringAccepted(record.csvString, formatHistoryDate(date) + '.csv', date);
          } catch (e) {
            showToast('Failed to load day: ' + e.message, 'error');
          }
        });
      });
    }

Note: showStatsPanel() is defined in Task 2. It is available at runtime since all JS is in the same script block.
